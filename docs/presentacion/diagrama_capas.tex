  \hd{Orquestador determinista}\\[2pt]

% ---- fila 3: 4 capas deterministas ----
% l2 y l3 se colocan a horcajadas del eje central (x=0): la cara este de l2

% las cuatro cajas tienen el mismo ancho, el grupo queda simétrico respecto
% al eje sin depender del inner sep. l1 y l4 encadenan hacia fuera.
\coordinate (drow) at ([yshift=-1.95cm]orq.south);
\node[layer, anchor=east] (l2) at ([xshift=-0.21cm]drow) {%
  \hd{Dependencias}\\[3pt]\sft{\emph{typosquatting}, scripts de instalación, URL/Git directas}};
\node[layer, anchor=west] (l3) at ([xshift=0.21cm]drow) {%
  \hd{Vulnerabilidades}\\[3pt]\sft{CVEs conocidas vía OSV en dependencias nuevas o actualizadas}};
\node[layer, anchor=east, left=0.42cm of l2] (l1) {%
  \hd{Estática}\\[3pt]\sft{Semgrep + YARA sobre los \mbox{\emph{hunks}} modificados}};
\node[layer, anchor=west, right=0.42cm of l3] (l4) {%
  \hd{Reputación}\\[3pt]\sft{identidad declarada frente al historial real de la cuenta}};

\node[fit=(l1)(l4), inner sep=10pt, draw=linestrong, line width=0.6pt,
      dash pattern=on 1.6pt off 1.6pt, rounded corners=3pt] (detbox) {};
\node[anchor=south west, font=\footnotesize\itshape, color=inksoft]
      at ([xshift=2pt,yshift=1pt]detbox.north west) {capas deterministas --- sin coste de tokens, no dependen del LLM};

% ---- fila 4: cortocircuito / puerta a la capa semántica ----
\node[gate, below=0.9cm of detbox.south] (gate) {%
  \hd{\textcolor{samber}{Cortocircuito}}\; ¿hace falta la capa semántica?\\[3pt]
  \sft{\textbf{Se omite la capa LLM} si las capas deterministas ya dan un veredicto claro,
  no queda presupuesto de \emph{tokens} o el \emph{diff} ya se analizó antes.}};

% ---- fila 5: capa semántica (opcional) ----
\node[optlayer, below=0.8cm of gate] (sem) {%
  \hd{Capa semántica (LLM + RAG local)}\; \textemdash\ \textbf{opcional}\\[3pt]
  \sft{Solo se ejecuta si el cortocircuito no ha resuelto. RAG sirve casos reales parecidos;
  el modelo puede invocar hasta 3 herramientas acotadas; salida JSON validada y cacheada.}};

% ---- fila 6: agregador ----
\node[band, below=0.8cm of sem] (agg) {%
  \hd{Agregador de \emph{scoring}}\; media ponderada renormalizada sobre las capas activas\\[3pt]
  \sft{Puntuación 0--100 explicada capa por capa. Clasifica cada hallazgo por naturaleza:
  \textcolor{semred}{$\blacksquare$ malicioso} \textperiodcentered\
  \textcolor{samber}{$\blacksquare$ vulnerabilidad} \textperiodcentered\
  \textcolor{inksoft}{$\blacksquare$ incertidumbre}. Bloqueo estricto si hay hallazgo malicioso
  de confianza alta/media. El resultado se persiste y alimenta el dashboard.}};

% ---- fila 7: semáforo ----
\node[band, below=0.8cm of agg] (out) {%
  \raisebox{-1pt}{\textcolor{semgreen}{$\bullet$}\,\textcolor{samber}{$\bullet$}\,\textcolor{semred}{$\bullet$}}\;

  \emph{check} de CI \textperiodcentered\ bloqueo de \emph{merge} configurable};
